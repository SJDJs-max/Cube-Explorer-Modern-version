cmake_minimum_required(VERSION 3.25)
project(CubeExplorerModern VERSION 0.3.2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(CUBE_BUILD_GUI "Build Qt user interface after core validation" OFF)
option(CUBE_BUILD_ENGINE "Build JSON engine" OFF)
include(CTest)

file(GLOB CORE_SOURCES CONFIGURE_DEPENDS src/core/*.cpp)
add_library(cube_core STATIC ${CORE_SOURCES})
target_include_directories(cube_core PUBLIC include)
target_compile_definitions(cube_core PUBLIC CUBE_PROJECT_ROOT="${CMAKE_CURRENT_SOURCE_DIR}")
if(MSVC)
  target_compile_options(cube_core PRIVATE /W4 /utf-8)
else()
  target_compile_options(cube_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
find_package(Threads REQUIRED)
target_link_libraries(cube_core PUBLIC Threads::Threads)

if(BUILD_TESTING)
  foreach(test_name model_tests twophase_tests optimal_tests pattern_tests pattern_model_cross_tests integration_tests template_search_tests user_regression_tests pattern_performance_tests mcc_tests)
    if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/${test_name}.cpp")
      add_executable(${test_name} tests/${test_name}.cpp)
      target_link_libraries(${test_name} PRIVATE cube_core)
      add_test(NAME ${test_name} COMMAND ${test_name})
      set_tests_properties(${test_name} PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
    endif()
  endforeach()
endif()

if(CUBE_BUILD_ENGINE OR CUBE_BUILD_GUI)
  find_package(Qt6 REQUIRED COMPONENTS Core)
endif()
if(CUBE_BUILD_ENGINE)
  add_subdirectory(src/engine)
endif()
if(CUBE_BUILD_GUI)
  find_package(Qt6 REQUIRED COMPONENTS Gui Widgets Concurrent PrintSupport)
  add_subdirectory(src/app)
endif()
